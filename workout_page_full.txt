ExerciseLibraryItem? findExercise(String id) {
  for (final exercise in exerciseLibrary) {
    if (exercise.id == id) {
      return exercise;
    }
  }

  return null;
}

String formatDuration(int seconds) {
  final hours = seconds ~/ 3600;
  final minutes = (seconds % 3600) ~/ 60;
  final remaining = seconds % 60;

  if (hours > 0) {
    return '${hours.toString().padLeft(2, '0')}:'
        '${minutes.toString().padLeft(2, '0')}:'
        '${remaining.toString().padLeft(2, '0')}';
  }

  return '${minutes.toString().padLeft(2, '0')}:'
      '${remaining.toString().padLeft(2, '0')}';
}

class WorkoutPage extends StatefulWidget {
  const WorkoutPage({super.key});

  @override
  State<WorkoutPage> createState() => _WorkoutPageState();
}

class _WorkoutPageState extends State<WorkoutPage> {
  List<Map<String, dynamic>> plans = [];
  bool loading = true;
  bool premium = false;

  @override
  void initState() {
    super.initState();
    _loadPlans();
  }

  Future<void> _loadPlans() async {
    final results = await Future.wait([
      FitnessStorage.getPlans(),
      FitnessStorage.isPremium(),
    ]);

    if (!mounted) return;

    setState(() {
      plans = List<Map<String, dynamic>>.from(results[0] as List);
      premium = results[1] as bool;
      loading = false;
    });
  }

  Future<void> _createPlan() async {
    final controller = TextEditingController();

    final name = await showDialog<String>(
      context: context,
      builder: (dialogContext) {
        return Directionality(
          textDirection: TextDirection.rtl,
          child: AlertDialog(
            backgroundColor: cardColor,
            title: const Text(
              'إنشاء جدول جديد',
              style: TextStyle(color: Colors.white),
            ),
            content: TextField(
              controller: controller,
              autofocus: true,
              style: const TextStyle(color: Colors.white),
              decoration: InputDecoration(
                hintText: 'مثال: Push',
                hintStyle: const TextStyle(color: mutedColor),
                labelText: 'اسم الجدول',
                labelStyle: const TextStyle(color: mutedColor),
                enabledBorder: OutlineInputBorder(
                  borderRadius: BorderRadius.circular(14),
                  borderSide: const BorderSide(
                    color: Color(0xFF29332F),
                  ),
                ),
                focusedBorder: OutlineInputBorder(
                  borderRadius: BorderRadius.circular(14),
                  borderSide: const BorderSide(
                    color: pharmacyGreen,
                  ),
                ),
              ),
            ),
            actions: [
              TextButton(
                onPressed: () => Navigator.pop(dialogContext),
                child: const Text('إلغاء'),
              ),
              FilledButton(
                style: FilledButton.styleFrom(
                  backgroundColor: pharmacyGreen,
                ),
                onPressed: () {
                  final value = controller.text.trim();

                  if (value.isEmpty) return;

                  Navigator.pop(dialogContext, value);
                },
                child: const Text('إنشاء'),
              ),
            ],
          ),
        );
      },
    );

    controller.dispose();

    if (!mounted || name == null || name.trim().isEmpty) {
      return;
    }

    final canAdd = await FitnessStorage.canAddPlan();

    if (!mounted) return;

    if (!canAdd) {
      await _showPlanLimit();
      return;
    }

    final success = await FitnessStorage.addPlan(
      name: name.trim(),
      exercises: [],
    );

    if (!mounted) return;

    if (success) {
      await _loadPlans();
    }
  }

  Future<void> _showPlanLimit() async {
    final limit = premium ? 6 : 3;

    await showDialog<void>(
      context: context,
      builder: (dialogContext) {
        return Directionality(
          textDirection: TextDirection.rtl,
          child: AlertDialog(
            backgroundColor: cardColor,
            title: const Text(
              'وصلت إلى الحد الأقصى',
              style: TextStyle(color: Colors.white),
            ),
            content: Text(
              premium
                  ? 'حساب Premium يسمح لك بإنشاء $limit جداول.'
                  : 'الحساب المجاني يسمح لك بإنشاء $limit جداول.\n'
                      'يمكن لاحقًا ترقية الحساب إلى Premium ليصبح الحد 6 جداول.',
              style: const TextStyle(
                color: Colors.white70,
                height: 1.6,
              ),
            ),
            actions: [
              TextButton(
                onPressed: () => Navigator.pop(dialogContext),
                child: const Text('حسنًا'),
              ),
            ],
          ),
        );
      },
    );
  }

  Future<void> _deletePlan(Map<String, dynamic> plan) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (dialogContext) {
        return Directionality(
          textDirection: TextDirection.rtl,
          child: AlertDialog(
            backgroundColor: cardColor,
            title: const Text(
              'حذف الجدول؟',
              style: TextStyle(color: Colors.white),
            ),
            content: Text(
              'سيتم حذف جدول "${plan['name']}".',
              style: const TextStyle(color: Colors.white70),
            ),
            actions: [
              TextButton(
                onPressed: () => Navigator.pop(dialogContext, false),
                child: const Text('إلغاء'),
              ),
              FilledButton(
                style: FilledButton.styleFrom(
                  backgroundColor: Colors.red,
                ),
                onPressed: () => Navigator.pop(dialogContext, true),
                child: const Text('حذف'),
              ),
            ],
          ),
        );
      },
    );

    if (!mounted || confirmed != true) return;

    await FitnessStorage.deletePlan(
      int.tryParse(plan['id'].toString()) ?? 0,
    );

    if (!mounted) return;

    await _loadPlans();
  }

  Future<void> _openPlan(Map<String, dynamic> plan) async {
    await Navigator.push(
      context,
      MaterialPageRoute(
        builder: (_) => WorkoutPlanPage(
          plan: plan,
        ),
      ),
    );

    if (!mounted) return;

    await _loadPlans();
  }

    final limit = premium ? 6 : 3;

    return Padding(
      padding: const EdgeInsets.fromLTRB(20, 22, 20, 12),
      child: Row(
        children: [
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                const Text(
                  'تماريني',
                  style: TextStyle(
                    color: Colors.white,
                    fontSize: 28,
                    fontWeight: FontWeight.w900,
                  ),
                ),
                const SizedBox(height: 5),
                Text(
                  '${plans.length} من $limit جداول',
                  style: const TextStyle(
                    color: mutedColor,
                    fontSize: 13,
                  ),
                ),
              ],
            ),
          ),
          Container(
            padding: const EdgeInsets.symmetric(
              horizontal: 11,
              vertical: 8,
            ),
            decoration: BoxDecoration(
              color: premium
                  ? pharmacyGreen.withValues(alpha: 0.15)
                  : Colors.white.withValues(alpha: 0.05),
              borderRadius: BorderRadius.circular(12),
              border: Border.all(
                color: premium
                    ? pharmacyGreen.withValues(alpha: 0.35)
                    : Colors.white.withValues(alpha: 0.08),
              ),
            ),
            child: Row(
              children: [
                Icon(
                  premium
                      ? Icons.workspace_premium
                      : Icons.person_outline,
                  size: 17,
                  color: premium ? pharmacyGreen : Colors.white70,
                ),
                const SizedBox(width: 6),
                Text(
                  premium ? 'Premium' : 'مجاني',
                  style: TextStyle(
                    color: premium ? pharmacyGreen : Colors.white70,
                    fontWeight: FontWeight.w800,
                    fontSize: 12,
                  ),
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }

class WorkoutPlanPage extends StatefulWidget {
  final Map<String, dynamic> plan;

  const WorkoutPlanPage({
    super.key,
    required this.plan,
  });

  @override
  State<WorkoutPlanPage> createState() => _WorkoutPlanPageState();
}

class _WorkoutPlanPageState extends State<WorkoutPlanPage> {
  late String planName;
  late int planId;
  List<Map<String, dynamic>> exercises = [];

  @override
  void initState() {
    super.initState();

    planId = int.tryParse(
          widget.plan['id'].toString(),
        ) ??
        0;

    planName = widget.plan['name']?.toString() ?? 'جدول';

    final raw = widget.plan['exercises'];

    if (raw is List) {
      exercises = raw
          .map(
            (item) => Map<String, dynamic>.from(item),
          )
          .toList();
    }
  }

  Future<void> _save() async {
    await FitnessStorage.updatePlan(
      planId,
      name: planName,
      exercises: exercises,
    );
  }

  Future<void> _renamePlan() async {
    final controller = TextEditingController(
      text: planName,
    );

    final value = await showDialog<String>(
      context: context,
      builder: (dialogContext) {
        return Directionality(
          textDirection: TextDirection.rtl,
          child: AlertDialog(
            backgroundColor: cardColor,
            title: const Text(
              'تعديل اسم الجدول',
              style: TextStyle(color: Colors.white),
            ),
            content: TextField(
              controller: controller,
              autofocus: true,
              style: const TextStyle(color: Colors.white),
              decoration: const InputDecoration(
                hintText: 'اسم الجدول',
                hintStyle: TextStyle(color: mutedColor),
              ),
            ),
            actions: [
              TextButton(
                onPressed: () => Navigator.pop(dialogContext),
                child: const Text('إلغاء'),
              ),
              FilledButton(
                style: FilledButton.styleFrom(
                  backgroundColor: pharmacyGreen,
                ),
                onPressed: () {
                  final text = controller.text.trim();

                  if (text.isNotEmpty) {
                    Navigator.pop(dialogContext, text);
                  }
                },
                child: const Text('حفظ'),
              ),
            ],
          ),
        );
      },
    );

    controller.dispose();

    if (!mounted || value == null) return;

    setState(() {
      planName = value;
    });

    await _save();
  }

  Future<void> _addExercise() async {
    final selected = await showModalBottomSheet<ExerciseLibraryItem>(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (_) => const ExerciseLibrarySheet(),
    );

    if (!mounted || selected == null) return;

    final config = await showDialog<Map<String, int>>(
      context: context,
      builder: (dialogContext) => ExerciseConfigDialog(
        exercise: selected,
      ),
    );

    if (!mounted || config == null) return;

    setState(() {
      exercises.add({
        'id': selected.id,
        'name': selected.name,
        'englishName': selected.englishName,
        'muscle': selected.muscle,
        'sets': config['sets'] ?? 3,
        'targetReps': config['reps'] ?? 10,
      });
    });

  }

  Future<void> _editExercise(int index) async {
    final exercise = exercises[index];

    final selected = findExercise(
      exercise['id']?.toString() ?? '',
    );

    if (selected == null) return;

    final config = await showDialog<Map<String, int>>(
      context: context,
      builder: (dialogContext) => ExerciseConfigDialog(
        exercise: selected,
        initialSets: int.tryParse(
              exercise['sets'].toString(),
            ) ??
            3,
        initialReps: int.tryParse(
              exercise['targetReps'].toString(),
            ) ??
            10,
      ),
    );

    if (!mounted || config == null) return;

    setState(() {
      exercises[index]['sets'] = config['sets'] ?? 3;
      exercises[index]['targetReps'] =
          config['reps'] ?? 10;
    });

    await _save();
  }

  Future<void> _removeExercise(int index) async {
    final exercise = exercises[index];

    final confirmed = await showDialog<bool>(
      context: context,
      builder: (dialogContext) {
        return Directionality(
          textDirection: TextDirection.rtl,
          child: AlertDialog(
            backgroundColor: cardColor,
            title: const Text(
              'حذف التمرين؟',
              style: TextStyle(color: Colors.white),
            ),
            content: Text(
              exercise['name']?.toString() ?? '',
              style: const TextStyle(color: Colors.white70),
            ),
            actions: [
              TextButton(
                onPressed: () =>
                    Navigator.pop(dialogContext, false),
                child: const Text('إلغاء'),
              ),
              FilledButton(
                style: FilledButton.styleFrom(
                  backgroundColor: Colors.red,
                ),
                onPressed: () =>
                    Navigator.pop(dialogContext, true),
                child: const Text('حذف'),
              ),
            ],
          ),
        );
      },
    );

    if (!mounted || confirmed != true) return;

    setState(() {
      exercises.removeAt(index);
    });

    await _save();
  }

class WorkoutSessionPage extends StatefulWidget {
  final int planId;
  final String planName;
  final List<Map<String, dynamic>> exercises;

  const WorkoutSessionPage({
    super.key,
    required this.planId,
    required this.planName,
    required this.exercises,
  });

  @override
  State<WorkoutSessionPage> createState() =>
      _WorkoutSessionPageState();
}

class _WorkoutSessionPageState
    extends State<WorkoutSessionPage> {
  late List<Map<String, dynamic>> sessionExercises;

  Timer? elapsedTimer;
  Timer? restTimer;

  DateTime? startedAt;

  int elapsedSeconds = 0;
  int restSeconds = 0;

  bool saving = false;
  bool finished = false;

  // ============================================================
  // الأداء السابق لكل مجموعة
  // المفتاح = اسم التمرين + رقم الجولة
  // ============================================================

  final Map<String, Map<String, dynamic>?>
      previousPerformance = {};

  bool loadingPrevious = true;

  @override
  void initState() {
    super.initState();

    startedAt = DateTime.now();

    sessionExercises = widget.exercises.map(
      (exercise) {
        final setCount = int.tryParse(
              exercise['sets'].toString(),
            ) ??
            1;

        final targetReps = int.tryParse(
              exercise['targetReps'].toString(),
            ) ??
            10;

        final sets = List.generate(
          setCount,
          (index) => {
            'set': index + 1,
            'weight': 0.0,
            'reps': targetReps,
            'completed': false,
          },
        );

        return {
          'id': exercise['id']?.toString() ?? '',
          'name':
              exercise['name']?.toString() ?? '',
          'englishName':
              exercise['englishName']?.toString() ?? '',
          'muscle':
              exercise['muscle']?.toString() ?? '',
          'targetReps': targetReps,
          'sets': sets,
        };
      },
    ).toList();

    _loadAllPreviousPerformance();

    elapsedTimer = Timer.periodic(
      const Duration(seconds: 1),
      (_) {
        if (!mounted || finished) return;

        setState(() {
          elapsedSeconds++;
        });
      },
    );
  }

  // ============================================================
  // تحميل الأداء السابق لكل الجولات
  // ============================================================

  Future<void> _loadAllPreviousPerformance() async {
    for (final exercise in sessionExercises) {
      final name =
          exercise['name']?.toString() ?? '';

      final sets = exercise['sets'];

      if (sets is! List) continue;

      for (final rawSet in sets) {
        if (rawSet is! Map) continue;

        final setNumber = _toInt(
          rawSet['set'],
        );

        if (name.trim().isEmpty || setNumber <= 0) {
          continue;
        }

        final key = _previousKey(
          name,
          setNumber,
        );

        try {
          final previous =
              await FitnessStorage.getLastSetPerformance(
            name,
            setNumber: setNumber,
          );

          previousPerformance[key] = previous;
        } catch (_) {
          previousPerformance[key] = null;
        }
      }
    }

    if (!mounted) return;

    setState(() {
      loadingPrevious = false;
    });
  }

  String _previousKey(
    String exerciseName,
    int setNumber,
  ) {
    return '${exerciseName.trim()}__$setNumber';
  }

  Map<String, dynamic>? _getPrevious(
    Map<String, dynamic> exercise,
    int setNumber,
  ) {
    final name =
        exercise['name']?.toString() ?? '';

    return previousPerformance[
      _previousKey(name, setNumber)
    ];
  }

  @override
  void dispose() {
    elapsedTimer?.cancel();
    restTimer?.cancel();
    _disposeControllers();
    super.dispose();
  }

  void _disposeControllers() {
    for (final exercise in sessionExercises) {
      final sets = exercise['sets'];

      if (sets is! List) continue;

      for (final rawSet in sets) {
        if (rawSet is Map) {
          final weightController =
              rawSet['_weightController'];

          final repsController =
              rawSet['_repsController'];

          if (weightController
              is TextEditingController) {
            weightController.dispose();
          }

          if (repsController
              is TextEditingController) {
            repsController.dispose();
          }
        }
      }
    }
  }

  TextEditingController _weightController(
    Map<String, dynamic> set,
  ) {
    final existing =
        set['_weightController'];

    if (existing is TextEditingController) {
      return existing;
    }

    final controller = TextEditingController(
      text: _formatNumber(
        _toDouble(set['weight']),
      ),
    );

    set['_weightController'] = controller;

    return controller;
  }

  TextEditingController _repsController(
    Map<String, dynamic> set,
  ) {
    final existing =
        set['_repsController'];

    if (existing is TextEditingController) {
      return existing;
    }

    final controller = TextEditingController(
      text: _toInt(set['reps']).toString(),
    );

    set['_repsController'] = controller;

    return controller;
  }

  double _toDouble(dynamic value) {
    if (value is double) return value;

    if (value is int) {
      return value.toDouble();
    }

    return double.tryParse(
          value?.toString() ?? '',
        ) ??
        0;
  }

  int _toInt(dynamic value) {
    if (value is int) return value;

    if (value is double) {
      return value.round();
    }

    return int.tryParse(
          value?.toString() ?? '',
        ) ??
        0;
  }

  String _formatNumber(double value) {
    if (value == 0) return '';

    if (value == value.roundToDouble()) {
      return value.toInt().toString();
    }

    return value.toString();
  }

  bool _allSetsCompleted() {
    for (final exercise in sessionExercises) {
      final sets = exercise['sets'];

      if (sets is! List) {
        return false;
      }

      for (final rawSet in sets) {
        if (rawSet['completed'] != true) {
          return false;
        }
      }
    }

    return true;
  }

  int _completedSetsCount() {
    var count = 0;

    for (final exercise in sessionExercises) {
      final sets = exercise['sets'];

      if (sets is! List) continue;

      for (final rawSet in sets) {
        if (rawSet['completed'] == true) {
          count++;
        }
      }
    }

    return count;
  }

  int _totalSetsCount() {
    var count = 0;

    for (final exercise in sessionExercises) {
      final sets = exercise['sets'];

      if (sets is List) {
        count += sets.length;
      }
    }

    return count;
  }

  Future<void> _loadPrevious(
    Map<String, dynamic> exercise,
    Map<String, dynamic> set,
  ) async {
    final name =
        exercise['name']?.toString() ?? '';

    final setNumber = _toInt(
      set['set'],
    );

    if (name.trim().isEmpty || setNumber <= 0) {
      return;
    }

    final key = _previousKey(
      name,
      setNumber,
    );

    Map<String, dynamic>? previous =
        previousPerformance[key];

    // إذا لم يتم تحميله سابقًا، نحاول جلبه الآن.
    if (previous == null && !previousPerformance.containsKey(key)) {
      previous =
          await FitnessStorage.getLastSetPerformance(
        name,
        setNumber: setNumber,
      );

      previousPerformance[key] = previous;
    }

    if (!mounted || previous == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'لا يوجد أداء سابق محفوظ لهذه الجولة.',
          ),
        ),
      );
      return;
    }

    final weight = _toDouble(
      previous['weight'],
    );

    final reps = _toInt(
      previous['reps'],
    );

    setState(() {
      set['weight'] = weight;
      set['reps'] = reps;

      final weightController =
          set['_weightController'];

      if (weightController
          is TextEditingController) {
        weightController.text =
            _formatNumber(weight);
      }

      final repsController =
          set['_repsController'];

      if (repsController
          is TextEditingController) {
        repsController.text =
            reps.toString();
      }
    });
  }

  void _startRest() {
    restTimer?.cancel();

    setState(() {
      restSeconds = 90;
    });

    restTimer = Timer.periodic(
      const Duration(seconds: 1),
      (timer) {
        if (!mounted) {
          timer.cancel();
          return;
        }

        if (restSeconds <= 1) {
          timer.cancel();

          setState(() {
            restSeconds = 0;
          });

          return;
        }

        setState(() {
          restSeconds--;
        });
      },
    );
  }

  void _skipRest() {
    restTimer?.cancel();

    setState(() {
      restSeconds = 0;
    });
  }

  Future<void> _completeSet(
    Map<String, dynamic> set,
  ) async {
    final weightController =
        _weightController(set);

    final repsController =
        _repsController(set);

    final weight =
        double.tryParse(
              weightController.text.trim(),
            ) ??
            0;

    final reps =
        int.tryParse(
              repsController.text.trim(),
            ) ??
            0;

    if (reps <= 0) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'أدخل عدد التكرارات أولاً.',
          ),
        ),
      );
      return;
    }

    // حفظ بيانات الجولة داخل نفس الخريطة الأصلية.
    setState(() {
      set['weight'] = weight;
      set['reps'] = reps;
      set['completed'] = true;
    });

    _startRest();

    if (_allSetsCompleted()) {
      await Future<void>.delayed(
        const Duration(milliseconds: 250),
      );

      if (!mounted) return;

      await _showFinishDialog();
    }
  }

  Future<void> _showFinishDialog() async {
    if (finished || saving) return;

    final shouldFinish = await showDialog<bool>(
      context: context,
      builder: (dialogContext) {
        return Directionality(

                color: Colors.white,
                fontWeight: FontWeight.w900,
              ),
            ),
            content: Text(
              'مدة التمرين: ${formatDuration(elapsedSeconds)}',
              style: const TextStyle(

              ),
            ),
            actions: [
              TextButton(
                onPressed: () =>
                    Navigator.pop(dialogContext, false),
                child: const Text(
                  'العودة للتمرين',

                style: FilledButton.styleFrom(
                  backgroundColor: pharmacyGreen,
                ),
                onPressed: () =>
                    Navigator.pop(dialogContext, true),
                child: const Text(
                  'إنهاء وحفظ',
                ),
              ),
            ],
          ),
        );
      },

      return;
    }

    await _finishWorkout();
  }

  // حفظ الجلسة بالكامل
  // ============================================================

  Future<void> _finishWorkout() async {
    if (saving) return;

    setState(() {
      saving = true;
      finished = true;
    });

    elapsedTimer?.cancel();
    restTimer?.cancel();

    final cleanExercises =
        sessionExercises.map(
      (exercise) {
        final rawSets = exercise['sets'];

        final cleanSets =
            <Map<String, dynamic>>[];

        if (rawSets is List) {
          for (final rawSet in rawSets) {
            if (rawSet is! Map) continue;

            cleanSets.add({
              'set': _toInt(
                rawSet['set'],
              ),
              'weight': _toDouble(
                rawSet['weight'],
              ),
              'reps': _toInt(
                rawSet['reps'],
              ),
              'completed':
                  rawSet['completed'] == true,
            });
          }
        }

        return {
          'exerciseId':
              exercise['id']?.toString() ?? '',
          'exercise':
              exercise['name']?.toString() ?? '',
          'sets': cleanSets,
        };
      },
    ).toList();

    await FitnessStorage.saveWorkoutSession(
      planId: widget.planId,
      planName: widget.planName,
      startedAt:
          startedAt ?? DateTime.now(),
      durationSeconds: elapsedSeconds,
      exercises: cleanExercises,
    );

    if (!mounted) return;

    Navigator.pop(context, true);
  }

  Future<bool> _onWillPop() async {
    if (finished || saving) {
      return false;
    }

    final leave = await showDialog<bool>(
      context: context,
      builder: (dialogContext) {
        return Directionality(
          textDirection: TextDirection.rtl,
          child: AlertDialog(
            backgroundColor: cardColor,
            title: const Text(
              'إنهاء التمرين؟',
              style: TextStyle(color: Colors.white),
            ),
            content: const Text(
              'لم يتم حفظ الجلسة الحالية بعد. هل تريد الخروج؟',
              style: TextStyle(
                color: Colors.white70,
              ),
            ),
            actions: [
              TextButton(
                onPressed: () =>
                    Navigator.pop(dialogContext, false),
                child: const Text('البقاء'),
              ),
              FilledButton(
                style: FilledButton.styleFrom(
                  backgroundColor: Colors.red,
                ),
                onPressed: () =>
                    Navigator.pop(dialogContext, true),
                child: const Text('خروج'),
              ),
            ],
          ),
        );
      },
    );

    return leave == true;
  }

  Widget _buildSetRow(
    Map<String, dynamic> exercise,
    Map<String, dynamic> set,
  ) {
    final completed =
        set['completed'] == true;

    final weightController =
        _weightController(set);

    final repsController =
        _repsController(set);

    final setNumber = _toInt(
      set['set'],
    );

    final previous =
        _getPrevious(
      exercise,
      setNumber,
    );

    final previousWeight = previous == null
        ? 0.0
        : _toDouble(previous['weight']);

    final previousReps = previous == null
        ? 0
        : _toInt(previous['reps']);

    final hasPrevious =
        previous != null &&
        (previousWeight > 0 || previousReps > 0);

    return Container(
      margin: const EdgeInsets.only(
        bottom: 10,
      ),
      padding: const EdgeInsets.fromLTRB(
        10,
        10,
        10,
        11,
      ),
      decoration: BoxDecoration(
        color: completed
            ? pharmacyGreen.withValues(alpha: 0.07)
            : Colors.white.withValues(alpha: 0.035),
        borderRadius: BorderRadius.circular(16),
        border: Border.all(
          color: completed
              ? pharmacyGreen.withValues(alpha: 0.25)
              : Colors.white.withValues(alpha: 0.05),
        ),
      ),
      child: Column(
        children: [
          Row(
            children: [
              Container(
                width: 35,
                height: 35,
                alignment: Alignment.center,
                decoration: BoxDecoration(
                  color: completed
                      ? pharmacyGreen
                      : Colors.white.withValues(
                          alpha: 0.06,
                        ),
                  shape: BoxShape.circle,
                ),
                child: completed
                    ? const Icon(
                        Icons.check,
                        color: Colors.white,
                        size: 19,
                      )
                    : Text(
                        '$setNumber',
                        style: const TextStyle(
                          color: Colors.white70,
                          fontWeight: FontWeight.w800,
                        ),
                      ),
              ),
              const SizedBox(width: 9),
              Expanded(
                child: _numberField(
                  controller: weightController,
                  label: 'الوزن',
                  suffix: 'كجم',
                  enabled: !completed,
                  onChanged: (value) {
                    set['weight'] =
                        double.tryParse(value) ?? 0;
                  },
                ),
              ),
              const SizedBox(width: 7),
              Expanded(
                child: _numberField(
                  controller: repsController,
                  label: 'التكرار',
                  suffix: 'عدة',
                  enabled: !completed,
                  onChanged: (value) {
                    set['reps'] =
                        int.tryParse(value) ?? 0;
                  },
                ),
              ),
              const SizedBox(width: 6),
              if (!completed)
                IconButton(
                  tooltip: 'إكمال الجولة',

                  icon: const Icon(
                    Icons.check_circle_outline,
                    color: pharmacyGreen,

                  width: 40,

            ],

          // الأداء السابق يظهر مباشرة تحت الجولة
          // ======================================================

            width: double.infinity,
            padding: const EdgeInsets.symmetric(
              horizontal: 10,
              vertical: 8,
            ),
            decoration: BoxDecoration(
              color: hasPrevious
                  ? pharmacyGreen.withValues(alpha: 0.055)
                  : Colors.white.withValues(alpha: 0.025),
              borderRadius: BorderRadius.circular(11),
            ),
            child: Row(
              children: [
                Icon(
                  Icons.history,
                  size: 15,
                  color: hasPrevious
                      ? pharmacyGreen
                      : Colors.white30,
                ),
                const SizedBox(width: 7),
                Expanded(
                  child: Text(
                    hasPrevious
                        ? 'السابق: ${_formatNumber(previousWeight)} كجم × $previousReps'
                        : loadingPrevious
                            ? 'جاري تحميل الأداء السابق...'
                            : 'لا يوجد أداء سابق لهذه الجولة',
                    style: TextStyle(
                      color: hasPrevious
                          ? Colors.white70

                      fontSize: 11,
                      fontWeight: hasPrevious
                          ? FontWeight.w700

                      exercise,
                      set,
                    ),
                    style: TextButton.styleFrom(
                      foregroundColor: pharmacyGreen,
                      padding:
                          const EdgeInsets.symmetric(
                        horizontal: 8,
                        vertical: 4,
                      ),
                      minimumSize: Size.zero,
                      tapTargetSize:
                          MaterialTapTargetSize.shrinkWrap,
                    ),
                    child: const Text(
                      'استخدام السابق',
                      style: TextStyle(
                        fontSize: 11,
                        fontWeight: FontWeight.w900,
                      ),
                    ),
                  ),
              ],
            ),
          ),
        ],
      ),
    );
  }
